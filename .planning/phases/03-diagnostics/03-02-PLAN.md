```c
void tquic_mib_seq_show(struct seq_file *seq)
{
    unsigned long sum[ARRAY_SIZE(tquic_snmp_list) - 1];
    struct net *net = seq_file_net(seq);
    int i;

    seq_puts(seq, "\nTquicExt:");
    for (i = 0; tquic_snmp_list[i].name; i++)
        seq_printf(seq, " %s", tquic_snmp_list[i].name);

    seq_puts(seq, "\nTquicExt:");
    memset(sum, 0, sizeof(sum));
    if (net->mib.tquic_statistics)
        snmp_get_cpu_field_batch(sum, tquic_snmp_list,
                                 net->mib.tquic_statistics);
    for (i = 0; tquic_snmp_list[i].name; i++)
        seq_printf(seq, " %lu", sum[i]);
    seq_putc(seq, '\n');
}
```

**Helper for EQUIC error -> MIB counter mapping**:
```c
static inline enum linux_tquic_mib_field tquic_equic_to_mib(u32 error_code)
{
    if (error_code < EQUIC_BASE || error_code >= EQUIC_BASE + 17)
        return TQUIC_MIB_NUM;  /* Invalid, no counter */
    return TQUIC_MIB_EQUIC_NO_ERROR + (error_code - EQUIC_BASE);
}
```
