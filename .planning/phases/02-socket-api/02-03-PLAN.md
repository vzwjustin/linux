```c
/*
 * TQUIC Stream ioctls
 *
 * Use ioctl on connection socket to create stream file descriptors.
 * Stream fds are first-class file descriptors supporting poll/epoll/select.
 *
 * Note: TQUIC uses a streams-only I/O model. sendmsg/recvmsg work on
 * stream sockets, not the connection socket. The connection socket is
 * used for control (connect, listen, accept, stream creation).
 */

/* ioctl magic number for TQUIC */
#define TQUIC_IOC_MAGIC  'Q'

/**
 * struct tquic_stream_args - Arguments for TQUIC_NEW_STREAM ioctl
 * @stream_id: OUT - Assigned stream ID after successful creation
 * @flags: IN - Stream type flags (TQUIC_STREAM_BIDI or TQUIC_STREAM_UNIDI)
 * @reserved: Reserved for future use, must be 0
 *
 * On success, the ioctl returns the new stream's file descriptor.
 * The stream_id field is populated with the assigned QUIC stream ID.
 */
struct tquic_stream_args {
    __u64 stream_id;        /* OUT: assigned stream ID */
    __u32 flags;            /* IN: TQUIC_STREAM_BIDI or TQUIC_STREAM_UNIDI */
    __u32 reserved;         /* Must be 0 */
};

/* Create new stream, returns fd on success */
#define TQUIC_NEW_STREAM    _IOWR(TQUIC_IOC_MAGIC, 1, struct tquic_stream_args)

/* Stream type flags */
#define TQUIC_STREAM_BIDI   0x00  /* Bidirectional stream (default) */
#define TQUIC_STREAM_UNIDI  0x01  /* Unidirectional stream (send-only) */

/* Stream limit sockopt (read-only) */
#define TQUIC_STREAMS_AVAILABLE  60  /* Get number of streams that can be opened */
```

The ioctl returns a file descriptor on success, negative errno on failure:
- -ENOTCONN: Connection not established
- -EAGAIN: Would block and O_NONBLOCK set (stream limit)
- -EINTR: Interrupted while waiting for stream credit
- -EINVAL: Invalid flags
- -ENOMEM: Memory allocation failed

```c
    TQUIC_STREAM_IDLE = 0,      /* Not yet created */

    TQUIC_STREAM_HALF_CLOSED_LOCAL,  /* We sent FIN */

    TQUIC_STREAM_CLOSED,        /* Fully closed */
```

```c
/*
 * TQUIC: Stream Socket Implementation
 *
 * Provides first-class stream file descriptors for QUIC stream multiplexing.
 * Each stream is a separate socket that can be used with poll/epoll/select.
 */

#include <linux/module.h>
#include <linux/net.h>
#include <linux/socket.h>
#include <linux/file.h>
#include <linux/poll.h>
#include <linux/anon_inodes.h>
#include <net/sock.h>
#include <net/tquic.h>
#include "protocol.h"

/*
 * Stream socket state
 * Stored in socket->sk_user_data to link stream fd to tquic_stream
 */
struct tquic_stream_sock {
    struct tquic_stream *stream;
    struct tquic_connection *conn;
    struct sock *parent_sk;      /* Connection socket */
    wait_queue_head_t wait;
};

/*
 * Stream management functions
 */

/**
 * tquic_stream_open - Create a new stream on connection
 */
struct tquic_stream *tquic_stream_open(struct tquic_connection *conn, bool is_bidi)
{
    struct tquic_stream *stream;
    u64 *next_id;

    if (!conn)
        return NULL;

    stream = kzalloc(sizeof(*stream), GFP_KERNEL);
    if (!stream)
        return NULL;

    /* Assign stream ID based on initiator role and direction */
    /* Client-initiated: even (bidi: 0,4,8..., uni: 2,6,10...)
     * Server-initiated: odd (bidi: 1,5,9..., uni: 3,7,11...) */
    if (is_bidi) {
        next_id = &conn->next_stream_id_bidi;
        stream->id = *next_id;
        *next_id += 4;
    } else {
        next_id = &conn->next_stream_id_uni;
        stream->id = *next_id;
        *next_id += 4;
    }

    stream->conn = conn;
    stream->state = TQUIC_STREAM_OPEN;
    stream->is_bidi = is_bidi;

    /* Initialize buffers */
    skb_queue_head_init(&stream->send_buf);
    skb_queue_head_init(&stream->recv_buf);

    /* Set initial flow control limits (from transport params) */
    stream->max_send_data = conn->peer_params.initial_max_stream_data_bidi_local;
    stream->max_recv_data = conn->local_params.initial_max_stream_data_bidi_remote;

    stream->send_offset = 0;
    stream->recv_offset = 0;
    stream->fin_received = false;
    stream->fin_sent = false;

    spin_lock_init(&stream->lock);

    /* Add to connection's stream list */
    spin_lock_bh(&conn->lock);
    list_add_tail(&stream->list, &conn->streams);
    conn->stream_count++;
    spin_unlock_bh(&conn->lock);

    pr_debug("tquic: opened stream id=%llu bidi=%d\n", stream->id, is_bidi);
    return stream;
}

/**
 * tquic_stream_close - Close and clean up a stream
 */
void tquic_stream_close(struct tquic_stream *stream)
{
    struct tquic_connection *conn;

    if (!stream)
        return;

    conn = stream->conn;

    spin_lock_bh(&stream->lock);
    stream->state = TQUIC_STREAM_CLOSED;
    spin_unlock_bh(&stream->lock);

    /* Purge buffers */
    skb_queue_purge(&stream->send_buf);
    skb_queue_purge(&stream->recv_buf);

    /* Remove from connection */
    if (conn) {
        spin_lock_bh(&conn->lock);
        list_del(&stream->list);
        conn->stream_count--;
        spin_unlock_bh(&conn->lock);
    }

    pr_debug("tquic: closed stream id=%llu\n", stream->id);
    kfree(stream);
}

/**
 * tquic_stream_find - Find stream by ID on connection
 */
struct tquic_stream *tquic_stream_find(struct tquic_connection *conn, u64 stream_id)
{
    struct tquic_stream *stream;

    if (!conn)
        return NULL;

    spin_lock_bh(&conn->lock);
    list_for_each_entry(stream, &conn->streams, list) {
        if (stream->id == stream_id) {
            spin_unlock_bh(&conn->lock);
            return stream;
        }
    }
    spin_unlock_bh(&conn->lock);

    return NULL;
}

/**
 * tquic_stream_socket_create - Create a new stream socket
 * @conn: Parent connection
 * @parent_sk: Parent socket (connection socket)
 * @flags: Stream type flags (TQUIC_STREAM_BIDI/UNIDI)
 * @stream_id: OUT - Assigned stream ID
 *
 * Returns: File descriptor on success, negative errno on failure
 */
int tquic_stream_socket_create(struct tquic_connection *conn,
                               struct sock *parent_sk,
                               u32 flags, u64 *stream_id)
{
    struct socket *sock;
    struct tquic_stream_sock *ss;
    struct tquic_stream *stream;
    int fd, err;

    /* Create socket structure */
    err = sock_create_kern(sock_net(parent_sk), parent_sk->sk_family,
                           SOCK_STREAM, 0, &sock);
    if (err < 0)
        return err;

    /* Override ops to stream-specific */
    sock->ops = &tquic_stream_ops;

    /* Allocate stream socket state */
    ss = kzalloc(sizeof(*ss), GFP_KERNEL);
    if (!ss) {
        sock_release(sock);
        return -ENOMEM;
    }

    /* Open stream on connection */
    stream = tquic_stream_open(conn, !(flags & TQUIC_STREAM_UNIDI));
    if (!stream) {
        kfree(ss);
        sock_release(sock);
        return -ENOMEM;
    }

    /* Link stream to socket */
    ss->stream = stream;
    ss->conn = conn;
    ss->parent_sk = parent_sk;
    init_waitqueue_head(&ss->wait);
    sock->sk->sk_user_data = ss;

    /* Get file descriptor */
    fd = sock_map_fd(sock, O_CLOEXEC);
    if (fd < 0) {
        tquic_stream_close(stream);
        kfree(ss);
        sock_release(sock);
        return fd;
    }

    *stream_id = stream->id;
    pr_debug("tquic: created stream socket, id=%llu fd=%d\n", stream->id, fd);

    return fd;
}

static int tquic_stream_release(struct socket *sock)
{
    struct tquic_stream_sock *ss;

    if (!sock->sk)
        return 0;

    ss = sock->sk->sk_user_data;
    if (ss) {
        if (ss->stream)
            tquic_stream_close(ss->stream);
        kfree(ss);
    }

    sock->sk->sk_user_data = NULL;
    return 0;
}

static int tquic_stream_sendmsg(struct socket *sock, struct msghdr *msg, size_t len)
{
    struct tquic_stream_sock *ss = sock->sk->sk_user_data;
    struct tquic_stream *stream;
    struct tquic_connection *conn;
    struct sk_buff *skb;
    size_t copied = 0;

    if (!ss || !ss->stream || !ss->conn)
        return -ENOTCONN;

    stream = ss->stream;
    conn = ss->conn;

    /* Check stream and connection state */
    if (stream->state == TQUIC_STREAM_CLOSED ||
        stream->state == TQUIC_STREAM_RESET_SENT)
        return -EPIPE;

    if (conn->state != TQUIC_CONN_CONNECTED)
        return -ENOTCONN;

    /* Copy data to stream send buffer */
    while (copied < len) {
        size_t chunk = min_t(size_t, len - copied, 1200);

        skb = alloc_skb(chunk, GFP_KERNEL);
        if (!skb)
            return copied > 0 ? copied : -ENOMEM;

        if (copy_from_iter(skb_put(skb, chunk), chunk, &msg->msg_iter) != chunk) {
            kfree_skb(skb);
            return copied > 0 ? copied : -EFAULT;
        }

        skb_queue_tail(&stream->send_buf, skb);
        stream->send_offset += chunk;
        copied += chunk;
    }

    /* Trigger transmission */
    tquic_output_flush(conn);

    return copied;
}

static int tquic_stream_recvmsg(struct socket *sock, struct msghdr *msg,
                                size_t len, int flags)
{
    struct tquic_stream_sock *ss = sock->sk->sk_user_data;
    struct tquic_stream *stream;
    struct sk_buff *skb;
    size_t copied = 0;
    int err;

    if (!ss || !ss->stream)
        return -ENOTCONN;

    stream = ss->stream;

    /* Wait for data if blocking */
    while (skb_queue_empty(&stream->recv_buf)) {
        if (stream->fin_received)
            return 0;  /* EOF */

        if (flags & MSG_DONTWAIT)
            return -EAGAIN;

        err = wait_event_interruptible(ss->wait,
                !skb_queue_empty(&stream->recv_buf) ||
                stream->fin_received ||
                stream->state == TQUIC_STREAM_CLOSED);
        if (err)
            return -EINTR;

        if (stream->state == TQUIC_STREAM_CLOSED)
            return -ECONNRESET;
    }

    /* Copy data from receive buffer */
    while (copied < len && !skb_queue_empty(&stream->recv_buf)) {
        size_t chunk;

        skb = skb_dequeue(&stream->recv_buf);
        if (!skb)
            break;

        chunk = min_t(size_t, len - copied, skb->len);

        if (copy_to_iter(skb->data, chunk, &msg->msg_iter) != chunk) {
            skb_queue_head(&stream->recv_buf, skb);
            return copied > 0 ? copied : -EFAULT;
        }

        copied += chunk;
        stream->recv_offset += chunk;

        if (chunk < skb->len) {
            skb_pull(skb, chunk);
            skb_queue_head(&stream->recv_buf, skb);
        } else {
            kfree_skb(skb);
        }
    }

    return copied;
}

static __poll_t tquic_stream_poll(struct file *file, struct socket *sock,
                                  poll_table *wait)
{
    struct tquic_stream_sock *ss = sock->sk->sk_user_data;
    struct tquic_stream *stream;
    __poll_t mask = 0;

    if (!ss || !ss->stream)
        return EPOLLERR;

    stream = ss->stream;

    poll_wait(file, &ss->wait, wait);

    /* Check for readable data */
    if (!skb_queue_empty(&stream->recv_buf) || stream->fin_received)
        mask |= EPOLLIN | EPOLLRDNORM;

    /* Check for writable (flow control permitting) */
    if (stream->send_offset < stream->max_send_data)
        mask |= EPOLLOUT | EPOLLWRNORM;

    /* Check for errors */
    if (stream->state == TQUIC_STREAM_CLOSED ||
        stream->state == TQUIC_STREAM_RESET_RECVD)
        mask |= EPOLLHUP;

    return mask;
}
```

```c
/*
 * ioctl handler for connection socket
 */
static int tquic_ioctl(struct socket *sock, unsigned int cmd, unsigned long arg)
{
    struct sock *sk = sock->sk;
    struct tquic_sock *tsk = tquic_sk(sk);
    struct tquic_connection *conn = tsk->conn;
    void __user *uarg = (void __user *)arg;

    switch (cmd) {
    case TQUIC_NEW_STREAM: {
        struct tquic_stream_args args;
        u64 stream_id;
        bool is_bidi;
        int ret;

        /* Must be connected */
        if (!conn || conn->state != TQUIC_CONN_CONNECTED)
            return -ENOTCONN;

        /* Copy args from userspace */
        if (copy_from_user(&args, uarg, sizeof(args)))
            return -EFAULT;

        /* Validate flags */
        if (args.flags > TQUIC_STREAM_UNIDI || args.reserved != 0)
            return -EINVAL;

        is_bidi = !(args.flags & TQUIC_STREAM_UNIDI);

        /* Block until stream credit available (per CONTEXT.md) */
        ret = tquic_wait_for_stream_credit(conn, is_bidi,
                                           sock->file->f_flags & O_NONBLOCK);
        if (ret < 0)
            return ret;

        /* Create stream socket */
        ret = tquic_stream_socket_create(conn, sk, args.flags, &stream_id);
        if (ret < 0)
            return ret;

        /* Copy stream ID back to userspace */
        args.stream_id = stream_id;
        if (copy_to_user(uarg, &args, sizeof(args))) {
            /* Close the fd we just created on error */
            /* Note: ksys_close(ret) would be needed but complex */
            return -EFAULT;
        }

        /* Return the file descriptor */
        return ret;
    }

    default:
        return -ENOIOCTLCMD;
    }
}

/**
 * tquic_wait_for_stream_credit - Wait until stream can be opened
 * @conn: Connection
 * @is_bidi: True for bidirectional, false for unidirectional
 * @nonblock: True if O_NONBLOCK set
 *
 * Per CONTEXT.md: ioctl blocks until peer grants more streams.
 * Returns 0 when stream can be opened, negative errno on failure.
 */
static int tquic_wait_for_stream_credit(struct tquic_connection *conn,
                                        bool is_bidi, bool nonblock)
{
    u64 *next_id, *max_streams;

    if (is_bidi) {
        next_id = &conn->next_stream_id_bidi;
        max_streams = &conn->max_streams_bidi;
    } else {
        next_id = &conn->next_stream_id_uni;
        max_streams = &conn->max_streams_uni;
    }

    /* Check if we can open a stream */
    if (*next_id < *max_streams * 4)  /* Stream IDs increment by 4 */
        return 0;

    /* Would block */
    if (nonblock)
        return -EAGAIN;

    /* Block waiting for MAX_STREAMS from peer */
    /* Use conn->sk's wait queue or a dedicated stream_wq */
    return wait_event_interruptible(conn->sk->sk_wq->wait,
                                    *next_id < *max_streams * 4 ||
                                    conn->state != TQUIC_CONN_CONNECTED);
}
```
