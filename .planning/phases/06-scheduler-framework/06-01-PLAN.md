3. Implement tquic_sched_set_default():
```c
int tquic_sched_set_default(struct net *net, const char *name)
{
    struct tquic_sched_ops *sched, *old;

    rcu_read_lock();
    sched = tquic_sched_find(name);
    if (!sched) {
        rcu_read_unlock();
        return -ENOENT;
    }
    if (!bpf_try_module_get(sched, sched->owner)) {
        rcu_read_unlock();
        return -EBUSY;
    }
    rcu_read_unlock();

    spin_lock(&tquic_sched_list_lock);
    old = xchg(&net->tquic.default_scheduler, sched);
    spin_unlock(&tquic_sched_list_lock);

    if (old)
        bpf_module_put(old, old->owner);

    return 0;
}
EXPORT_SYMBOL_GPL(tquic_sched_set_default);
```

4. Implement tquic_sched_init_conn() with connection state check:
```c
int tquic_sched_init_conn(struct tquic_connection *conn, const char *name)
{
    struct tquic_sched_ops *sched;
    struct net *net;

    /* Scheduler can only be set before connection established */
    if (conn->state != TQUIC_CONN_IDLE)
        return -EISCONN;

    if (name && name[0]) {
        rcu_read_lock();
        sched = tquic_sched_find(name);
        rcu_read_unlock();
    } else {
        /* Use per-netns default */
        net = sock_net(conn->sk);
        rcu_read_lock();
        sched = tquic_get_default_sched(net);
        if (!sched)
            sched = &tquic_sched_aggregate; /* fallback */
        rcu_read_unlock();
    }

    if (!sched)
        return -ENOENT;

    if (!bpf_try_module_get(sched, sched->owner))
        return -EBUSY;

    conn->scheduler = sched;
    conn->sched_priv = NULL;

    if (sched->init)
        sched->init(conn);

    return 0;
}
EXPORT_SYMBOL_GPL(tquic_sched_init_conn);
```

```c
int tquic_sched_get_path(struct tquic_connection *conn,
                         struct tquic_sched_path_result *result,
                         u32 flags)
{
    struct tquic_sched_ops *sched;

    if (!conn || !result)
        return -EINVAL;

    memset(result, 0, sizeof(*result));

    rcu_read_lock();
    sched = conn->scheduler;
    if (!sched || !sched->get_path) {
        rcu_read_unlock();
        return -ENOENT;
    }

    int ret = sched->get_path(conn, result, flags);
    rcu_read_unlock();

    return ret;
}
```

```c
static char tquic_sched_name_buf[TQUIC_SCHED_NAME_MAX] = "aggregate";

static int proc_tquic_scheduler(struct ctl_table *table, int write,
                                void *buffer, size_t *lenp, loff_t *ppos)
{
    struct net *net = container_of(table->data, struct net,
                                   tquic.sched_name);
    char name[TQUIC_SCHED_NAME_MAX];
    int ret;

    if (!write) {
        /* Read current default */
        const char *current = tquic_sched_get_default(net);
        if (!current)
            current = "aggregate";
        strscpy(name, current, sizeof(name));
        table->data = name;
        ret = proc_dostring(table, write, buffer, lenp, ppos);
        table->data = &net->tquic.sched_name;
        return ret;
    }

    /* Write new default */
    ret = proc_dostring(table, write, buffer, lenp, ppos);
    if (ret)
        return ret;

    /* Validate and set */
    strscpy(name, net->tquic.sched_name, sizeof(name));
    ret = tquic_sched_set_default(net, name);
    if (ret)
        pr_warn("TQUIC: invalid scheduler '%s'\n", name);

    return ret;
}
```
