```c
// SPDX-License-Identifier: GPL-2.0
/*
 * TQUIC BLEST Scheduler (Blocking Estimation-based)
 *
 * Estimates blocking delay for each path and waits for fast path
 * if it will complete sooner than slow path send time.
 *
 * Algorithm (from IFIP 2016 paper):
 * - Track inflight data per path
 * - Estimate completion time = (inflight + new_data) / bandwidth
 * - If fast path completion < slow path send time, wait for fast path
 * - Reduces head-of-line blocking at receiver
 *
 * Reference: "BLEST: Blocking estimation-based MPTCP scheduler for
 * heterogeneous networks" (IFIP Networking 2016)
 */
```

2. BLEST private state:
```c
struct blest_path_state {
    u64 inflight_bytes;         /* Bytes in flight on this path */
    u64 last_send_time_us;      /* Timestamp of last send */
    u64 send_rate;              /* Estimated send rate (bytes/s) */
    u32 rtt_us;                 /* Path RTT */
};

struct blest_sched_data {
    struct blest_path_state paths[TQUIC_MAX_PATHS];
    u32 segment_size;           /* Default segment size for estimation */
};

#define BLEST_DEFAULT_SEGMENT_SIZE  1200
```

```c
static int blest_get_path_idx(struct tquic_connection *conn,
                              struct tquic_path *target)
{
    struct tquic_path *path;
    int idx = 0;

    list_for_each_entry_rcu(path, &conn->paths, list) {
        if (path == target)
            return idx;
        idx++;
        if (idx >= TQUIC_MAX_PATHS)
            break;
    }
    return -1;
}
```

4. Blocking estimation calculation:
```c
/*
 * Calculate blocking time if we send on slow_path
 *
 * Blocking occurs when:
 * - We send on slow path
 * - Fast path has inflight data that will arrive first
 * - Slow path packet blocks reassembly
 *
 * Blocking time = (slow_send_time + slow_rtt) - (fast_complete_time)
 */
static s64 blest_blocking_estimate(struct blest_sched_data *sd,
                                   int slow_idx, int fast_idx,
                                   u32 segment_size)
{
    struct blest_path_state *slow = &sd->paths[slow_idx];
    struct blest_path_state *fast = &sd->paths[fast_idx];
    u64 slow_arrival_time;
    u64 fast_completion_time;

    if (fast->send_rate == 0 || slow->rtt_us == 0)
        return 0;

    /* Slow path: time for new segment to arrive at receiver */
    slow_arrival_time = slow->rtt_us;

    /* Fast path: time for inflight to complete */
    fast_completion_time = (fast->inflight_bytes * 1000000ULL) / fast->send_rate;
    fast_completion_time += fast->rtt_us;

    /* If slow arrives before fast completes: blocking */
    if (slow_arrival_time < fast_completion_time)
        return (s64)(fast_completion_time - slow_arrival_time);

    return 0;
}
```

5. BLEST get_path implementation:
```c
static int blest_get_path(struct tquic_connection *conn,
                          struct tquic_sched_path_result *result,
                          u32 flags)
{
    struct blest_sched_data *sd = conn->sched_priv;
    struct tquic_path *path, *best = NULL;
    int best_idx = -1;
    int fast_idx = -1;
    u32 min_rtt = U32_MAX;
    s64 min_blocking = S64_MAX;
    int idx = 0;

    if (!sd)
        return -EINVAL;

    rcu_read_lock();

    /* First pass: find fastest path (min RTT) and update state */
    list_for_each_entry_rcu(path, &conn->paths, list) {
        if (idx >= TQUIC_MAX_PATHS)
            break;

        if (path->state == TQUIC_PATH_ACTIVE) {
            struct blest_path_state *ps = &sd->paths[idx];

            /* Sync from path stats */
            ps->rtt_us = path->stats.rtt_smoothed;
            if (ps->rtt_us == 0)
                ps->rtt_us = 100000;  /* 100ms default */

            ps->send_rate = path->stats.bandwidth;
            if (ps->send_rate == 0)
                ps->send_rate = 125000;  /* 1 Mbps default */

            if (ps->rtt_us < min_rtt) {
                min_rtt = ps->rtt_us;
                fast_idx = idx;
            }
        }
        idx++;
    }

    if (fast_idx < 0) {
        rcu_read_unlock();
        return -ENOENT;
    }

    /* Second pass: find path with minimum blocking estimate */
    idx = 0;
    list_for_each_entry_rcu(path, &conn->paths, list) {
        if (idx >= TQUIC_MAX_PATHS)
            break;

        if (path->state == TQUIC_PATH_ACTIVE) {
            s64 blocking;

            if (idx == fast_idx) {
                /* Fast path: no blocking by definition */
                blocking = 0;
            } else {
                blocking = blest_blocking_estimate(sd, idx, fast_idx,
                                                   sd->segment_size);
            }

            if (blocking < min_blocking) {
                min_blocking = blocking;
                best = path;
                best_idx = idx;
            }
        }
        idx++;
    }

    if (!best) {
        rcu_read_unlock();
        return -ENOENT;
    }

    /* Update inflight for selected path */
    sd->paths[best_idx].inflight_bytes += sd->segment_size;
    sd->paths[best_idx].last_send_time_us = ktime_get_ns() / 1000;

    result->primary = best;
    result->backup = NULL;
    result->flags = 0;

    rcu_read_unlock();
    return 0;
}
```

```c
static void blest_ack_received(struct tquic_connection *conn,
                               struct tquic_path *path,
                               u64 acked_bytes)
{
    struct blest_sched_data *sd = conn->sched_priv;
    int idx;

    if (!sd)
        return;

    idx = blest_get_path_idx(conn, path);
    if (idx >= 0 && idx < TQUIC_MAX_PATHS) {
        struct blest_path_state *ps = &sd->paths[idx];
        if (ps->inflight_bytes >= acked_bytes)
            ps->inflight_bytes -= acked_bytes;

            ps->inflight_bytes = 0;

}
```

1. License and documentation:
```c
// SPDX-License-Identifier: GPL-2.0
/*
 * TQUIC ECF Scheduler (Earliest Completion First)
 *
 * Selects the path with the earliest estimated completion time for
 * the next segment.
 *
 * Algorithm (from ACM SIGMETRICS 2017):
 * - For each path: est_completion = (inflight + segment_size) / send_rate + RTT
 * - Choose path with minimum est_completion
 * - Considers both bandwidth and RTT (not just RTT like MinRTT)
 * - Better utilization under path heterogeneity
 *
 * Reference: "ECF: An MPTCP Path Scheduler to Manage Heterogeneous Paths"
 * (ACM SIGMETRICS 2017)
 */
```

2. ECF private state:
```c
struct ecf_path_state {
    u64 inflight_bytes;         /* Bytes in flight */
    u64 send_rate;              /* Send rate (bytes/s) from delivery rate */
    u32 rtt_us;                 /* RTT in microseconds */
    u64 delivered;              /* Total delivered bytes (for rate) */
    u64 delivered_time_us;      /* Time of delivered sample */
};

struct ecf_sched_data {
    struct ecf_path_state paths[TQUIC_MAX_PATHS];
    u32 segment_size;
};

#define ECF_DEFAULT_SEGMENT_SIZE    1200
#define ECF_RATE_SAMPLE_INTERVAL_US 10000   /* 10ms rate sample */
```

3. Completion time calculation:
```c
/*
 * Calculate estimated completion time for a segment on this path
 *
 * completion_time = (inflight + segment_size) / send_rate + RTT
 *
 * Returns time in microseconds
 */
static u64 ecf_completion_time(struct ecf_path_state *ps, u32 segment_size)
{
    u64 queue_time;
    u64 propagation_time;

    if (ps->send_rate == 0)
        return U64_MAX;  /* Unknown rate, avoid this path */

    /* Time for inflight + new segment to be transmitted */
    queue_time = ((ps->inflight_bytes + segment_size) * 1000000ULL) /
                 ps->send_rate;

    /* Add RTT for propagation */
    propagation_time = ps->rtt_us;

    return queue_time + propagation_time;
}
```

4. Rate estimation update:
```c
static void ecf_update_rate(struct ecf_sched_data *sd, int idx,
                            struct tquic_path *path)
{
    struct ecf_path_state *ps = &sd->paths[idx];
    u64 now_us = ktime_get_ns() / 1000;
    u64 delivered_now = path->stats.tx_bytes;  /* Approximation */

    /* Update RTT from path stats */
    ps->rtt_us = path->stats.rtt_smoothed;
    if (ps->rtt_us == 0)
        ps->rtt_us = 100000;  /* 100ms default */

    /* Calculate delivery rate if we have a previous sample */
    if (ps->delivered_time_us > 0) {
        u64 interval = now_us - ps->delivered_time_us;
        u64 delivered = delivered_now - ps->delivered;

        if (interval >= ECF_RATE_SAMPLE_INTERVAL_US && delivered > 0) {
            /* rate = delivered / interval, convert to bytes/s */
            ps->send_rate = (delivered * 1000000ULL) / interval;
            ps->delivered = delivered_now;
            ps->delivered_time_us = now_us;
        }
    } else {
        /* First sample */
        ps->delivered = delivered_now;
        ps->delivered_time_us = now_us;
        /* Use bandwidth estimate if available */
        ps->send_rate = path->stats.bandwidth;
        if (ps->send_rate == 0)
            ps->send_rate = 125000;  /* 1 Mbps default */
    }
}
```

5. ECF get_path implementation:
```c
static int ecf_get_path(struct tquic_connection *conn,
                        struct tquic_sched_path_result *result,
                        u32 flags)
{
    struct ecf_sched_data *sd = conn->sched_priv;
    struct tquic_path *path, *best = NULL;
    u64 min_completion = U64_MAX;
    int best_idx = -1;
    int idx = 0;

    if (!sd)
        return -EINVAL;

    rcu_read_lock();

    list_for_each_entry_rcu(path, &conn->paths, list) {
        u64 completion;

        if (idx >= TQUIC_MAX_PATHS)
            break;

        if (path->state != TQUIC_PATH_ACTIVE) {
            idx++;
            continue;
        }

        /* Update rate estimate */
        ecf_update_rate(sd, idx, path);

        /* Calculate completion time */
        completion = ecf_completion_time(&sd->paths[idx], sd->segment_size);

        if (completion < min_completion) {
            min_completion = completion;
            best = path;
            best_idx = idx;
        }

        idx++;
    }

    if (!best) {
        rcu_read_unlock();
        return -ENOENT;
    }

    /* Update inflight for selected path */
    sd->paths[best_idx].inflight_bytes += sd->segment_size;

    result->primary = best;
    result->backup = NULL;
    result->flags = 0;

    rcu_read_unlock();
    return 0;
}
```
