```c
static int do_tquic_setsockopt(struct sock *sk, int level, int optname,
                               sockptr_t optval, unsigned int optlen)
{
    struct tquic_sock *tsk = tquic_sk(sk);
    int ret = 0;

    if (level != IPPROTO_TQUIC)
        return -ENOPROTOOPT;

    switch (optname) {
    case SO_TQUIC_SCHEDULER: {
        char name[TQUIC_SCHED_NAME_MAX];

        if (optlen < 1 || optlen >= TQUIC_SCHED_NAME_MAX)
            return -EINVAL;

        if (copy_from_sockptr(name, optval, optlen))
            return -EFAULT;
        name[optlen] = '\0';

        /* Must be called before connect/listen */
        lock_sock(sk);
        if (tsk->conn && tsk->conn->state != TQUIC_CONN_IDLE) {
            ret = -EISCONN;
        } else if (tsk->conn) {
            ret = tquic_sched_init_conn(tsk->conn, name);
        } else {
            /* Store for later when connection is created */
            strscpy(tsk->requested_scheduler, name,
                    sizeof(tsk->requested_scheduler));
        }
        release_sock(sk);
        return ret;
    }

    /* Other options... */
    default:
        return -ENOPROTOOPT;
    }
}
```

3. Add getsockopt handler:
```c
static int do_tquic_getsockopt(struct sock *sk, int level, int optname,
                               char __user *optval, int __user *optlen)
{
    struct tquic_sock *tsk = tquic_sk(sk);
    int len;
    int ret = 0;

    if (level != IPPROTO_TQUIC)
        return -ENOPROTOOPT;

    if (get_user(len, optlen))
        return -EFAULT;

    switch (optname) {
    case SO_TQUIC_SCHEDULER: {
        const char *name;
        int name_len;

        lock_sock(sk);
        if (tsk->conn && tsk->conn->scheduler) {
            struct tquic_sched_ops *sched = tsk->conn->scheduler;
            name = sched->name;
        } else if (tsk->requested_scheduler[0]) {
            name = tsk->requested_scheduler;
        } else {
            /* Return default */
            name = tquic_sched_get_default(sock_net(sk));
            if (!name)
                name = "aggregate";
        }
        release_sock(sk);

        name_len = strlen(name) + 1;
        if (len < name_len)
            return -EINVAL;

        if (copy_to_user(optval, name, name_len))
            return -EFAULT;
        if (put_user(name_len, optlen))
            return -EFAULT;

        return 0;
    }

    default:
        return -ENOPROTOOPT;
    }
}
```
