   /*
    * Enable coupled CC for a connection
    * This switches all paths to use coordinated CWND management
    */
   int tquic_cong_enable_coupling(struct tquic_connection *conn,
                                  enum tquic_coupled_algo algo)
   {
       struct tquic_coupled_state *cs;
       struct tquic_path *path;

       if (conn->coupled_enabled)
           return 0;  /* Already enabled */

       /* Create coupled state for connection */
       cs = tquic_coupled_create(conn, algo);
       if (!cs)
           return -ENOMEM;

       /* Attach all existing paths to coupled state */
       spin_lock(&conn->paths_lock);
       list_for_each_entry(path, &conn->paths, list) {
           if (path->state == TQUIC_PATH_ACTIVE ||
               path->state == TQUIC_PATH_VALIDATED) {
               tquic_coupled_attach_path(cs, path);
           }
       }
       spin_unlock(&conn->paths_lock);

       conn->coupled_cc = cs;
       conn->coupled_enabled = true;
       conn->coupled_algo = algo;

       pr_info("tquic: enabled coupled CC (%s) for connection\n",
               algo == TQUIC_COUPLED_OLIA ? "OLIA" :
               algo == TQUIC_COUPLED_BALIA ? "BALIA" : "LIA");

       return 0;
   }

   /*
    * Disable coupled CC
    */
   void tquic_cong_disable_coupling(struct tquic_connection *conn)
   {
       if (!conn->coupled_enabled)
           return;

       tquic_coupled_destroy(conn->coupled_cc);
       conn->coupled_cc = NULL;
       conn->coupled_enabled = false;
   }
   ```

   struct coupled_state *tquic_coupled_create(struct tquic_connection *conn,
                                              enum tquic_coupled_algo algo)
   {
       struct coupled_state *state;

       state = kzalloc(sizeof(*state), GFP_KERNEL);
       if (!state)
           return NULL;

       state->conn = conn;
       state->algo = algo;  /* OLIA is default per RESEARCH.md */
       state->global_alpha = COUPLED_SCALE;
       state->use_cubic = true;  /* Default: CUBIC integration */
       state->use_bbr = false;

       INIT_LIST_HEAD(&state->subflows);
       spin_lock_init(&state->lock);

       state->sbd.check_interval = 1000;  /* 1 second */

       pr_debug("tquic_coupled: created for connection (algo: %s)\n",
                algo == COUPLED_ALGO_OLIA ? "OLIA" :
                algo == COUPLED_ALGO_BALIA ? "BALIA" : "LIA");

       return state;
   }

   void tquic_coupled_destroy(struct coupled_state *state)
   {
       struct coupled_subflow *sf, *tmp;

       if (!state)
           return;

       spin_lock(&state->lock);
       list_for_each_entry_safe(sf, tmp, &state->subflows, list) {
           list_del(&sf->list);
           kfree(sf);
       }
       spin_unlock(&state->lock);

       kfree(state->sbd.corr_matrix);
       kfree(state);
   }
   ```

2. Add path attach/detach functions:
   ```c
   /*
    * Attach a path to coupled CC state
    */
   int tquic_coupled_attach_path(struct coupled_state *state,
                                 struct tquic_path *path)
   {
       struct coupled_subflow *sf;

       if (!state || !path)
           return -EINVAL;

       /* Check if already attached */
       spin_lock(&state->lock);
       list_for_each_entry(sf, &state->subflows, list) {
           if (sf->path_id == path->path_id) {
               spin_unlock(&state->lock);
               return 0;  /* Already attached */
           }
       }
       spin_unlock(&state->lock);

       /* Create new subflow */
       sf = kzalloc(sizeof(*sf), GFP_ATOMIC);
       if (!sf)
           return -ENOMEM;

       sf->path = path;
       sf->path_id = path->path_id;
       sf->cwnd = COUPLED_INIT_CWND;
       sf->ssthresh = COUPLED_MAX_CWND;
       sf->in_slow_start = true;
       sf->alpha = COUPLED_SCALE;
       sf->cubic_w_max = sf->cwnd;

       spin_lock(&state->lock);
       list_add_tail(&sf->list, &state->subflows);
       state->num_subflows++;
       spin_unlock(&state->lock);

       /* Update global alpha */
       coupled_update_alpha(state);

       pr_debug("tquic_coupled: attached path %u (total: %u)\n",
                path->path_id, state->num_subflows);

       return 0;
   }

   void tquic_coupled_detach_path(struct coupled_state *state, u32 path_id)
   {
       struct coupled_subflow *sf, *tmp;

       if (!state)
           return;

       spin_lock(&state->lock);
       list_for_each_entry_safe(sf, tmp, &state->subflows, list) {
           if (sf->path_id == path_id) {
               list_del(&sf->list);
               state->num_subflows--;
               kfree(sf);
               break;
           }
       }
       spin_unlock(&state->lock);

       /* Update global alpha without this path */
       if (state->num_subflows > 0)
           coupled_update_alpha(state);
   }
   ```

   ```c
   static void tquic_process_ack_ecn(struct tquic_connection *conn,
                                     struct tquic_path *path,
                                     u64 ect0, u64 ect1, u64 ce)
   {
       struct net *net = sock_net(conn->sk);
       static u64 last_ce_count[TQUIC_MAX_PATHS];

       /* Check if ECN is enabled */
       if (!net->tquic.ecn_enabled)
           return;

       /* Check for new CE marks */
       if (ce > last_ce_count[path->path_id]) {
           u64 new_ce = ce - last_ce_count[path->path_id];

           pr_debug("tquic: ECN CE marks detected on path %u: %llu new\n",
                    path->path_id, new_ce);

           /* Signal congestion via on_ecn callback */
           tquic_cong_on_ecn(path, new_ce);

           last_ce_count[path->path_id] = ce;
       }
   }
   ```

3. Add tquic_cong_on_ecn() dispatch in tquic_cong.c:
   ```c
   void tquic_cong_on_ecn(struct tquic_path *path, u64 ecn_ce_count)
   {
       struct tquic_connection *conn = path->conn;

       if (conn && conn->coupled_enabled && conn->coupled_cc) {
           /* ECN in coupled mode: treat as loss signal */
           tquic_coupled_on_ecn(conn->coupled_cc, path, ecn_ce_count);
           return;
       }

       if (path->cong_ops && path->cong_ops->on_ecn)
           path->cong_ops->on_ecn(path->cong, ecn_ce_count);
       else if (path->cong_ops && path->cong_ops->on_loss) {
           /* Fallback: treat ECN CE as equivalent to loss
            * This is conservative but correct behavior
            * RFC 3168: React to CE marks as you would to loss
            */
           path->cong_ops->on_loss(path->cong, 1200);  /* 1 MSS */
       }

       if (path->cong_ops && path->cong_ops->get_cwnd)
           path->stats.cwnd = path->cong_ops->get_cwnd(path->cong);
   }
   ```
