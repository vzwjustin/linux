1. Add pacing rate accessor to tquic_cong.c:
   ```c
   /*
    * Get pacing rate for a path (bytes/sec)
    *
    * Uses bandwidth estimation when available (BBR-style),
    * falls back to cwnd/RTT approximation.
    */
   u64 tquic_cong_get_pacing_rate(struct tquic_path *path)
   {
       u64 rate = 0;

       /* First try CC-provided pacing rate (bandwidth-based) */
       if (path->cong_ops && path->cong_ops->get_pacing_rate)
           rate = path->cong_ops->get_pacing_rate(path->cong);

       /* Fallback: cwnd / RTT approximation */
       if (rate == 0 && path->stats.rtt_smoothed > 0) {
           u64 cwnd = path->stats.cwnd ?: 10 * 1200;
           /* rate = cwnd / RTT (in bytes/sec) */
           rate = cwnd * USEC_PER_SEC / path->stats.rtt_smoothed;
       }

       /* Minimum pacing rate: 1 MSS per 10ms = 120KB/s */
       if (rate < 120000)
           rate = 120000;

       return rate;
   }
   ```

2. Add socket pacing rate update in tquic_output.c:
   ```c
   /*
    * Update socket pacing rate from CC state
    *
    * This integrates with FQ qdisc when available.
    * If FQ is attached to the interface, it will pace packets
    * according to sk->sk_pacing_rate. Otherwise, we fall back
    * to internal pacing.
    */
   static void tquic_update_pacing(struct sock *sk, struct tquic_path *path)
   {
       u64 pacing_rate;
       struct tquic_sock *tsk = tquic_sk(sk);
       struct net *net = sock_net(sk);

       /* Check if pacing is enabled */
       if (!net->tquic.pacing_enabled)
           return;

       pacing_rate = tquic_cong_get_pacing_rate(path);

       /*
        * Update socket pacing rate for FQ qdisc integration.
        * FQ checks sk->sk_pacing_rate to pace packets.
        * If FQ is not configured, this has no effect (internal pacing needed).
        */
       WRITE_ONCE(sk->sk_pacing_rate, pacing_rate);

       /*
        * Check pacing status:
        * SK_PACING_NONE   - No pacing active
        * SK_PACING_NEEDED - Internal pacing required (no FQ)
        * SK_PACING_FQ     - FQ qdisc handles pacing
        */
       if (smp_load_acquire(&sk->sk_pacing_status) == SK_PACING_NEEDED) {
           /* Internal pacing needed - FQ not available */
           tquic_internal_pacing_update(tsk, pacing_rate);
       }
   }

   /*
    * Internal pacing when FQ qdisc not available
    * Uses EDT (Earliest Departure Time) timestamps
    */
   static void tquic_internal_pacing_update(struct tquic_sock *tsk, u64 rate)
   {
       /* Store rate for use in tquic_pacing_send() */
       tsk->pacing_rate = rate;
   }
   ```

3. Add pacing check before packet send:

   ```c
   /*
    * Check if we should pace before sending
    * Returns number of bytes we can send now, or 0 if pacing
    */
   static bool tquic_pacing_allows_send(struct sock *sk, struct sk_buff *skb)
   {
       /* If FQ is handling pacing, always allow (FQ will pace) */
       if (smp_load_acquire(&sk->sk_pacing_status) == SK_PACING_FQ)
           return true;

       /* Check internal pacing timer */
       /* FQ qdisc integration: set skb->tstamp for EDT */
       if (sk->sk_pacing_rate > 0) {
           /* Calculate departure time based on pacing rate */
           u64 len_ns = div64_u64((u64)skb->len * NSEC_PER_SEC,
                                   sk->sk_pacing_rate);

           /* Set EDT timestamp for FQ */
           skb->tstamp = tcp_wstamp_ns(sk) + len_ns;
       }

       return true;  /* Allow send, FQ or internal timer handles pacing */
   }
   ```

2. Add loss tracking to tquic_cong_on_loss():
   ```c
   void tquic_cong_on_loss(struct tquic_path *path, u64 bytes_lost)
   {
       struct tquic_connection *conn = path->conn;
       static struct tquic_loss_tracker trackers[TQUIC_MAX_PATHS];
       struct tquic_loss_tracker *tracker;

       if (!path || path->path_id >= TQUIC_MAX_PATHS)
           return;

       tracker = &trackers[path->path_id];

       /* Check if this is in the same round */
       if (path->stats.tx_packets > tracker->current_delivered + path->stats.cwnd / 1200) {
           /* New round - reset counter */
           tracker->consecutive_losses = 0;
           tracker->round_delivered = path->stats.tx_packets;
       }
       tracker->current_delivered = path->stats.tx_packets;

       /* Count this loss */
       tracker->consecutive_losses++;

       pr_debug("tquic: loss on path %u, consecutive=%u\n",
                path->path_id, tracker->consecutive_losses);

       /* Check for degradation threshold */
       if (tracker->consecutive_losses >= TQUIC_PATH_DEGRADE_LOSS_THRESHOLD) {
           pr_warn("tquic: path %u degraded after %u consecutive losses\n",
                   path->path_id, tracker->consecutive_losses);

           /* Signal path manager for degradation */
           tquic_bond_path_failed(conn, path);

           /* Reset counter after degradation */
           tracker->consecutive_losses = 0;
       }

       /* Call CC's on_loss */
       if (conn && conn->coupled_enabled && conn->coupled_cc) {
           tquic_coupled_on_loss(conn->coupled_cc, path, bytes_lost);
       } else if (path->cong_ops && path->cong_ops->on_loss) {
           path->cong_ops->on_loss(path->cong, bytes_lost);
       }

       /* Update path stats */
       if (path->cong_ops && path->cong_ops->get_cwnd)
           path->stats.cwnd = path->cong_ops->get_cwnd(path->cong);
   }
   ```
