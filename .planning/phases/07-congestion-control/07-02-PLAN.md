1. Add proc_tquic_cc_algorithm() handler (similar to proc_tquic_scheduler):
   ```c
   static int proc_tquic_cc_algorithm(struct ctl_table *table, int write,
                                      void *buffer, size_t *lenp, loff_t *ppos)
   {
       struct net *net = current->nsproxy->net_ns;
       char name[NETNS_TQUIC_CC_NAME_MAX];
       struct ctl_table tmp_table;
       int ret;

       if (!write) {
           /* Read current default CC for this netns */
           const char *current_name;

           rcu_read_lock();
           if (net->tquic.default_cong)
               current_name = net->tquic.default_cong->name;
           else
               current_name = "cubic";
           rcu_read_unlock();

           strscpy(name, current_name, sizeof(name));

           memset(&tmp_table, 0, sizeof(tmp_table));
           tmp_table.procname = table->procname;
           tmp_table.data = name;
           tmp_table.maxlen = sizeof(name);
           tmp_table.mode = table->mode;

           return proc_dostring(&tmp_table, write, buffer, lenp, ppos);
       }

       /* Write: get new CC name from user */
       strscpy(name, net->tquic.cc_name, sizeof(name));

       memset(&tmp_table, 0, sizeof(tmp_table));
       tmp_table.procname = table->procname;
       tmp_table.data = name;
       tmp_table.maxlen = sizeof(name);
       tmp_table.mode = table->mode;

       ret = proc_dostring(&tmp_table, write, buffer, lenp, ppos);
       if (ret)
           return ret;

       /* Validate CC exists */
       ret = tquic_cong_set_default(net, name);
       if (ret) {
           pr_warn("tquic: unknown congestion control '%s'\n", name);
           return ret;
       }

       pr_debug("tquic: netns cc_algorithm set to '%s'\n", name);
       return 0;
   }
   ```

3. Add sockopt handler in tquic_socket.c setsockopt function:
   ```c
   case SO_TQUIC_CONGESTION:
       /* Only allowed before connect (same as scheduler) */
       if (tsk->conn && tsk->conn->state != TQUIC_CONN_IDLE)
           return -EISCONN;

       if (optlen > TQUIC_MAX_CC_NAME)
           return -EINVAL;

       if (copy_from_user(tsk->requested_congestion, optval, optlen))
           return -EFAULT;

       tsk->requested_congestion[optlen] = '\0';

       /* Validate CC exists */
       rcu_read_lock();
       if (!tquic_cong_find(tsk->requested_congestion)) {
           rcu_read_unlock();
           tsk->requested_congestion[0] = '\0';
           return -ENOENT;
       }
       rcu_read_unlock();
       return 0;

   case SO_TQUIC_COUPLED:
       if (optlen < sizeof(int))
           return -EINVAL;
       if (copy_from_user(&val, optval, sizeof(int)))
           return -EFAULT;
       tsk->coupled_requested = !!val;
       return 0;
   ```

5. Add BBR auto-selection logic in tquic_cong_init_path():
   ```c
   int tquic_cong_init_path(struct tquic_path *path, const char *name)
   {
       struct tquic_connection *conn = path->conn;
       struct net *net = sock_net(conn->sk);
       struct tquic_cong_ops *ops;
       const char *algo_name = name;
       void *state;

       /* If no specific algorithm requested */
       if (!algo_name || !algo_name[0]) {
           /* Check for BBR auto-selection on high-RTT paths */
           u32 threshold = tquic_net_get_bbr_rtt_threshold(net);
           if (threshold > 0 && path->stats.rtt_smoothed >= threshold * 1000) {
               /* RTT >= threshold, auto-select BBR */
               algo_name = "bbr";
               pr_debug("tquic: auto-selecting BBR for path %u (RTT=%u us >= %u ms)\n",
                       path->path_id, path->stats.rtt_smoothed, threshold);
           } else {
               /* Use per-netns default */
               algo_name = tquic_net_get_cc_algorithm(net);
           }
       }

       rcu_read_lock();
       ops = tquic_cong_find(algo_name);
       if (!ops) {
           /* Fallback to cubic */
           ops = tquic_cong_find("cubic");
       }
       rcu_read_unlock();

       if (!ops)
           return -ENOENT;

       state = ops->init(path);
       if (!state)
           return -ENOMEM;

       path->cong = state;
       path->cong_ops = ops;

       return 0;
   }
   ```
