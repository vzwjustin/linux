<objective>
Implement the adaptive RB-tree reorder buffer for handling packet reordering across heterogeneous latency paths.

Purpose: The reorder buffer is the data plane for bandwidth aggregation. When packets arrive out of order (fast path delivers packet N+5 while slow path's packet N is still in flight), the buffer holds them until the gap is filled. This enables true aggregation across paths with vastly different latencies (fiber ~20ms + satellite ~600ms = 580ms spread).

Output: tquic_reorder.h/c with RB-tree based out-of-order queue (following MPTCP's pattern), O(log n) insertion, adaptive sizing based on RTT spread, gap timeout for stuck packets, memory limit enforcement.
</objective>

```c
    spin_lock_init(&rb->lock);
```

4. tquic_reorder_destroy():
```c
void tquic_reorder_destroy(struct tquic_reorder_buffer *rb)
{
    struct rb_node *p, *next;
    struct sk_buff *skb;

    if (!rb)
        return;

    cancel_delayed_work_sync(&rb->timeout_work);

    spin_lock_bh(&rb->lock);

    /* Free all buffered packets */
    for (p = rb_first(&rb->queue); p; p = next) {
        next = rb_next(p);
        skb = rb_to_skb(p);
        rb_erase(p, &rb->queue);
        kfree_skb(skb);
    }

    spin_unlock_bh(&rb->lock);

    pr_debug("reorder buffer destroyed, delivered=%llu timeout=%llu\n",
             rb->stats.packets_delivered, rb->stats.gap_timeouts);

    kfree(rb);
}
EXPORT_SYMBOL_GPL(tquic_reorder_destroy);
```

5. tquic_reorder_insert() - O(log n) RB-tree insertion:
```c
int tquic_reorder_insert(struct tquic_reorder_buffer *rb,
                         struct sk_buff *skb, u64 seq)
{
    struct rb_node **p = &rb->queue.rb_node;
    struct rb_node *parent = NULL;
    struct tquic_skb_cb *cb;
    int ret = 0;

    /* Set up control block */
    cb = TQUIC_SKB_CB(skb);
    cb->seq = seq;
    cb->end_seq = seq + skb->len;
    cb->recv_time = ktime_get();

    spin_lock_bh(&rb->lock);

    /* Check if this is the next expected packet */
    if (seq == rb->next_expected_seq) {
        /* In-order: deliver immediately */
        spin_unlock_bh(&rb->lock);
        /* Caller delivers to stream */
        return 1;  /* 1 = deliver now, don't buffer */
    }

    /* Old/duplicate packet? */
    if (before64(cb->end_seq, rb->next_expected_seq)) {
        spin_unlock_bh(&rb->lock);
        rb->stats.duplicates_dropped++;
        kfree_skb(skb);
        return 0;
    }

    /* Memory limit check */
    if (rb->buffer_bytes + skb->truesize > rb->max_buffer_bytes) {
        spin_unlock_bh(&rb->lock);
        rb->stats.backpressure_events++;
        return -ENOBUFS;  /* Apply backpressure */
    }

    /* Fast path: append to tail if RB tree empty or seq > last */
    if (RB_EMPTY_ROOT(&rb->queue)) {
        rb_link_node(&skb->rbnode, NULL, p);
        rb_insert_color(&skb->rbnode, &rb->queue);
        rb->last_skb = skb;
        rb->oldest_packet_time = cb->recv_time;
        goto inserted;
    }

    /* Fast path: append after last_skb */
    if (rb->last_skb && after64(seq, TQUIC_SKB_CB(rb->last_skb)->seq)) {
        p = &rb->last_skb->rbnode.rb_right;
        parent = &rb->last_skb->rbnode;
        rb_link_node(&skb->rbnode, parent, p);
        rb_insert_color(&skb->rbnode, &rb->queue);
        rb->last_skb = skb;
        goto inserted;
    }

    /* Standard RB-tree insertion */
    while (*p) {
        struct sk_buff *skb1 = rb_to_skb(*p);
        u64 seq1 = TQUIC_SKB_CB(skb1)->seq;
        parent = *p;

        if (before64(seq, seq1))
            p = &(*p)->rb_left;
        else if (after64(seq, seq1))
            p = &(*p)->rb_right;
        else {
            /* Duplicate */
            spin_unlock_bh(&rb->lock);
            rb->stats.duplicates_dropped++;
            kfree_skb(skb);
            return 0;
        }
    }

    rb_link_node(&skb->rbnode, parent, p);
    rb_insert_color(&skb->rbnode, &rb->queue);

inserted:
    rb->buffer_bytes += skb->truesize;
    atomic_inc(&rb->packet_count);
    if (after64(cb->end_seq, rb->max_buffered_seq))
        rb->max_buffered_seq = cb->end_seq;
    rb->stats.packets_inserted++;

    spin_unlock_bh(&rb->lock);

    /* Start gap timeout if not already running */
    if (!delayed_work_pending(&rb->timeout_work))
        queue_delayed_work(rb->wq, &rb->timeout_work,
                           msecs_to_jiffies(rb->gap_timeout_ms));

    return 0;  /* 0 = buffered, not delivered */
}
EXPORT_SYMBOL_GPL(tquic_reorder_insert);
```

6. tquic_reorder_drain() - deliver in-order packets:
```c
int tquic_reorder_drain(struct tquic_reorder_buffer *rb)
{
    struct rb_node *p;
    struct sk_buff *skb;
    int delivered = 0;

    spin_lock_bh(&rb->lock);

    p = rb_first(&rb->queue);
    while (p) {
        skb = rb_to_skb(p);
        u64 seq = TQUIC_SKB_CB(skb)->seq;

        if (after64(seq, rb->next_expected_seq))
            break;  /* Gap - stop draining */

        p = rb_next(p);
        rb_erase(&skb->rbnode, &rb->queue);
        rb->buffer_bytes -= skb->truesize;
        atomic_dec(&rb->packet_count);

        /* Update last_skb if we removed it */
        if (skb == rb->last_skb)
            rb->last_skb = p ? rb_to_skb(p) : NULL;

        spin_unlock_bh(&rb->lock);

        /* Deliver to stream reassembly */
        /* TODO: tquic_stream_deliver(rb->sk, skb); */
        rb->next_expected_seq = TQUIC_SKB_CB(skb)->end_seq;
        delivered++;
        rb->stats.packets_delivered++;

        spin_lock_bh(&rb->lock);
        p = rb_first(&rb->queue);  /* Re-fetch after unlock */
    }

    /* Update oldest packet time */
    if (p) {
        rb->oldest_packet_time = TQUIC_SKB_CB(rb_to_skb(p))->recv_time;
    }

    spin_unlock_bh(&rb->lock);

    return delivered;
}
EXPORT_SYMBOL_GPL(tquic_reorder_drain);
```

7. Gap timeout handler:
```c
static void tquic_reorder_timeout_handler(struct work_struct *work)
{
    struct delayed_work *dwork = to_delayed_work(work);
    struct tquic_reorder_buffer *rb = container_of(dwork,
                                        struct tquic_reorder_buffer,
                                        timeout_work);
    struct rb_node *p;
    ktime_t now = ktime_get();
    ktime_t threshold;
    u64 gap_start;

    threshold = ktime_sub_ms(now, rb->gap_timeout_ms);

    spin_lock_bh(&rb->lock);

    p = rb_first(&rb->queue);
    if (!p) {
        spin_unlock_bh(&rb->lock);
        return;
    }

    /* Check if oldest packet has timed out */
    if (ktime_before(rb->oldest_packet_time, threshold)) {
        struct sk_buff *first_skb = rb_to_skb(p);
        gap_start = rb->next_expected_seq;

        pr_debug("gap timeout at seq %llu, jumping to %llu\n",
                 gap_start, TQUIC_SKB_CB(first_skb)->seq);

        /* Jump next_expected_seq to first buffered packet */
        rb->next_expected_seq = TQUIC_SKB_CB(first_skb)->seq;
        rb->stats.gap_timeouts++;

        spin_unlock_bh(&rb->lock);

        /* Request retransmit for gap (Phase 05-03 wires this) */
        /* tquic_request_retransmit(rb->sk, gap_start, rb->next_expected_seq); */

        /* Drain now-deliverable packets */
        tquic_reorder_drain(rb);

        spin_lock_bh(&rb->lock);
    }

    /* Reschedule if buffer not empty */
    if (!RB_EMPTY_ROOT(&rb->queue)) {
        spin_unlock_bh(&rb->lock);
        queue_delayed_work(rb->wq, &rb->timeout_work,
                           msecs_to_jiffies(rb->gap_timeout_ms));
        return;
    }

    spin_unlock_bh(&rb->lock);
}
```

8. Helper functions:
```c
void tquic_reorder_update_timeout(struct tquic_reorder_buffer *rb,
                                   u32 min_rtt_us, u32 max_rtt_us)
{
    u32 spread_ms;

    if (max_rtt_us <= min_rtt_us)
        return;

    /* gap_timeout = 2 * rtt_spread + margin */
    spread_ms = (max_rtt_us - min_rtt_us) / 1000;
    rb->gap_timeout_ms = spread_ms * 2 + TQUIC_GAP_TIMEOUT_MARGIN_MS;

    pr_debug("gap timeout updated to %u ms (spread=%u ms)\n",
             rb->gap_timeout_ms, spread_ms);
}
EXPORT_SYMBOL_GPL(tquic_reorder_update_timeout);

size_t tquic_reorder_calc_size(u32 min_rtt_us, u32 max_rtt_us, u64 bandwidth)
{
    u32 spread_us;
    size_t size;

    if (max_rtt_us <= min_rtt_us || bandwidth == 0)
        return TQUIC_DEFAULT_REORDER_BUFFER;

    /* size = spread_ms * bandwidth_bytes_per_ms * 2 (safety) */
    spread_us = max_rtt_us - min_rtt_us;
    size = (spread_us / 1000) * (bandwidth / 1000) * 2;

    return clamp(size, (size_t)TQUIC_MIN_REORDER_BUFFER,
                       (size_t)TQUIC_MAX_REORDER_BUFFER);
}
EXPORT_SYMBOL_GPL(tquic_reorder_calc_size);

bool tquic_reorder_has_gap(struct tquic_reorder_buffer *rb)
{
    bool has_gap;
    spin_lock_bh(&rb->lock);
    has_gap = !RB_EMPTY_ROOT(&rb->queue);
    spin_unlock_bh(&rb->lock);
    return has_gap;
}
EXPORT_SYMBOL_GPL(tquic_reorder_has_gap);

size_t tquic_reorder_buffered_bytes(struct tquic_reorder_buffer *rb)
{
    size_t bytes;
    spin_lock_bh(&rb->lock);
    bytes = rb->buffer_bytes;
    spin_unlock_bh(&rb->lock);
    return bytes;
}
EXPORT_SYMBOL_GPL(tquic_reorder_buffered_bytes);
```

```c
void tquic_bonding_update_reorder_timeout(struct tquic_bonding_ctx *bc)
{
    struct tquic_path *path;
    u32 min_rtt = U32_MAX, max_rtt = 0;

    if (!bc->reorder || !bc->pm)
        return;

    rcu_read_lock();
    list_for_each_entry_rcu(path, &bc->pm->path_list, list) {
        if (path->state != TQUIC_PATH_ACTIVE &&
            path->state != TQUIC_PATH_VALIDATED)
            continue;

        if (path->metrics.srtt > 0) {
            min_rtt = min(min_rtt, path->metrics.srtt);
            max_rtt = max(max_rtt, path->metrics.srtt);
        }
    }
    rcu_read_unlock();

    if (min_rtt != U32_MAX && max_rtt > min_rtt)
        tquic_reorder_update_timeout(bc->reorder, min_rtt, max_rtt);
```
