<objective>
Implement seamless failover with zero application-visible packet loss on path failure.

Purpose: Failover is the reliability guarantee of bandwidth aggregation. When a path fails, all unacknowledged packets sent on that path must be retransmitted on remaining paths. The application must see a continuous stream with no gaps. This is what makes multi-path bonding truly seamless vs just multi-homing.

Output: tquic_failover.h/c with sent packet tracking, retransmit queue management, priority scheduling of failover packets, path failure detection (3x SRTT), and receiver-side deduplication.

```c
static u32 tquic_sent_packet_hash(const void *data, u32 len, u32 seed)
{
    const struct tquic_sent_packet *pkt = data;
    return jhash_2words((u32)pkt->packet_number,
                        (u32)(pkt->packet_number >> 32), seed);
}

static int tquic_sent_packet_cmp(struct rhashtable_compare_arg *arg,
                                  const void *obj)
{
    const struct tquic_sent_packet *pkt = obj;
    const u64 *pn = arg->key;
    return pkt->packet_number != *pn;
}

const struct rhashtable_params tquic_sent_packet_params = {
    .key_offset = offsetof(struct tquic_sent_packet, packet_number),
    .key_len = sizeof(u64),
    .head_offset = offsetof(struct tquic_sent_packet, hash),
    .hashfn = tquic_sent_packet_hash,
    .obj_cmpfn = tquic_sent_packet_cmp,
};
EXPORT_SYMBOL_GPL(tquic_sent_packet_params);
```

```c
    int err;

    fc = kzalloc(sizeof(*fc), gfp);

    INIT_LIST_HEAD(&fc->sent_packets);
    spin_lock_init(&fc->sent_lock);

    spin_lock_init(&fc->retransmit_lock);

    if (err) {
        kfree(fc);
        return NULL;

    bitmap_zero(fc->recv_bitmap, 256);
    fc->bonding = bc;

    pr_debug("failover context initialized\n");
    return fc;
}
```

4. tquic_failover_destroy():
```c
static void tquic_sent_packet_free(void *ptr, void *arg)
{
    struct tquic_sent_packet *pkt = ptr;
    if (pkt->skb)
        kfree_skb(pkt->skb);
    kfree(pkt);
}

void tquic_failover_destroy(struct tquic_failover_ctx *fc)
{
    struct tquic_sent_packet *pkt, *tmp;

    if (!fc)
        return;

    /* Free sent packets */
    rhashtable_free_and_destroy(&fc->sent_hash, tquic_sent_packet_free, NULL);

    /* Free retransmit queue */
    spin_lock_bh(&fc->retransmit_lock);
    list_for_each_entry_safe(pkt, tmp, &fc->retransmit_queue, list) {
        list_del(&pkt->list);
        if (pkt->skb)
            kfree_skb(pkt->skb);
        kfree(pkt);
    }
    spin_unlock_bh(&fc->retransmit_lock);

    pr_debug("failover destroyed: acked=%llu lost=%llu failovers=%llu\n",
             fc->stats.packets_acked, fc->stats.packets_lost,
             fc->stats.failover_events);

    kfree(fc);
}
EXPORT_SYMBOL_GPL(tquic_failover_destroy);
```

5. tquic_failover_track_sent():
```c
int tquic_failover_track_sent(struct tquic_failover_ctx *fc,
                              u64 packet_number, u8 path_id,
                              struct sk_buff *skb, u32 srtt_us)
{
    struct tquic_sent_packet *pkt;
    int err;

    pkt = kzalloc(sizeof(*pkt), GFP_ATOMIC);
    if (!pkt)
        return -ENOMEM;

    pkt->packet_number = packet_number;
    pkt->path_id = path_id;
    pkt->bytes = skb->len;
    pkt->ack_eliciting = true;
    pkt->sent_time = ktime_get();
    pkt->timeout_us = srtt_us * TQUIC_FAILOVER_TIMEOUT_MULT;
    pkt->skb = skb_clone(skb, GFP_ATOMIC);
    pkt->retransmit_count = 0;

    INIT_LIST_HEAD(&pkt->list);

    spin_lock_bh(&fc->sent_lock);

    err = rhashtable_insert_fast(&fc->sent_hash, &pkt->hash,
                                  tquic_sent_packet_params);
    if (err) {
        spin_unlock_bh(&fc->sent_lock);
        if (pkt->skb)
            kfree_skb(pkt->skb);
        kfree(pkt);
        return err;
    }

    list_add_tail(&pkt->list, &fc->sent_packets);
    fc->sent_count++;
    fc->stats.packets_tracked++;

    spin_unlock_bh(&fc->sent_lock);

    return 0;
}
EXPORT_SYMBOL_GPL(tquic_failover_track_sent);
```

6. tquic_failover_on_ack():
```c
void tquic_failover_on_ack(struct tquic_failover_ctx *fc,
                           u64 packet_number, u8 path_id)
{
    struct tquic_sent_packet *pkt;

    spin_lock_bh(&fc->sent_lock);

    pkt = rhashtable_lookup_fast(&fc->sent_hash, &packet_number,
                                  tquic_sent_packet_params);
    if (!pkt) {
        spin_unlock_bh(&fc->sent_lock);
        return;
    }

    /* Update highest ACKed for this path */
    if (packet_number > fc->highest_acked[path_id])
        fc->highest_acked[path_id] = packet_number;
    fc->last_ack_time[path_id] = ktime_get();

    /* Remove from tracking */
    rhashtable_remove_fast(&fc->sent_hash, &pkt->hash,
                           tquic_sent_packet_params);
    list_del(&pkt->list);
    fc->sent_count--;
    fc->stats.packets_acked++;

    spin_unlock_bh(&fc->sent_lock);

    if (pkt->skb)
        kfree_skb(pkt->skb);
    kfree(pkt);
}
EXPORT_SYMBOL_GPL(tquic_failover_on_ack);
```

7. tquic_failover_on_path_failed() - THE CRITICAL FUNCTION:
```c
void tquic_failover_on_path_failed(struct tquic_failover_ctx *fc, u8 path_id)
{
    struct tquic_sent_packet *pkt, *tmp;
    LIST_HEAD(requeue);
    int count = 0;

    pr_info("path %u failed, requeueing unacked packets\n", path_id);

    spin_lock_bh(&fc->sent_lock);

    /* Collect unacked packets from failed path */
    list_for_each_entry_safe(pkt, tmp, &fc->sent_packets, list) {
        if (pkt->path_id == path_id && pkt->ack_eliciting) {
            /* Remove from sent tracking */
            rhashtable_remove_fast(&fc->sent_hash, &pkt->hash,
                                   tquic_sent_packet_params);
            list_del(&pkt->list);
            fc->sent_count--;

            /* Move to temporary list */
            list_add_tail(&pkt->list, &requeue);
            count++;
        }
    }

    spin_unlock_bh(&fc->sent_lock);

    if (list_empty(&requeue))
        return;

    /* Add to retransmit queue (priority over new data) */
    spin_lock_bh(&fc->retransmit_lock);

    /* Enforce queue limit */
    while (fc->retransmit_count + count > TQUIC_MAX_RETRANSMIT_QUEUE &&
           !list_empty(&requeue)) {
        pkt = list_first_entry(&requeue, struct tquic_sent_packet, list);
        list_del(&pkt->list);
        if (pkt->skb)
            kfree_skb(pkt->skb);
        kfree(pkt);
        count--;
        fc->stats.packets_lost++;
    }

    /* Prepend to retransmit queue (oldest first = highest priority) */
    list_splice(&requeue, &fc->retransmit_queue);
    fc->retransmit_count += count;

    spin_unlock_bh(&fc->retransmit_lock);

    fc->stats.failover_events++;

    pr_info("requeued %d packets from path %u for retransmit\n", count, path_id);
}
EXPORT_SYMBOL_GPL(tquic_failover_on_path_failed);
```

8. tquic_failover_get_next() - for scheduler integration:
```c
struct sk_buff *tquic_failover_get_next(struct tquic_failover_ctx *fc)
{
    struct tquic_sent_packet *pkt;
    struct sk_buff *skb = NULL;

    spin_lock_bh(&fc->retransmit_lock);

    if (list_empty(&fc->retransmit_queue)) {
        spin_unlock_bh(&fc->retransmit_lock);
        return NULL;
    }

    pkt = list_first_entry(&fc->retransmit_queue,
                           struct tquic_sent_packet, list);
    list_del(&pkt->list);
    fc->retransmit_count--;

    if (pkt->skb) {
        skb = skb_clone(pkt->skb, GFP_ATOMIC);
        pkt->retransmit_count++;
        fc->stats.retransmits_sent++;
    }

    spin_unlock_bh(&fc->retransmit_lock);

    /* Free packet tracking (will be re-tracked when sent) */
    if (pkt->skb)
        kfree_skb(pkt->skb);
    kfree(pkt);

    return skb;
}
EXPORT_SYMBOL_GPL(tquic_failover_get_next);

bool tquic_failover_has_pending(struct tquic_failover_ctx *fc)
{
    bool has;
    spin_lock_bh(&fc->retransmit_lock);
    has = !list_empty(&fc->retransmit_queue);
    spin_unlock_bh(&fc->retransmit_lock);
    return has;
}
EXPORT_SYMBOL_GPL(tquic_failover_has_pending);
```

```c
bool tquic_failover_check_path_timeout(struct tquic_failover_ctx *fc,
                                        u8 path_id, u32 srtt_us)
{
    ktime_t now = ktime_get();
    ktime_t threshold;
    u32 timeout_us = srtt_us * TQUIC_FAILOVER_TIMEOUT_MULT;

    threshold = ktime_sub_us(now, timeout_us);

    return ktime_before(fc->last_ack_time[path_id], threshold);
}
```

10. Receiver-side deduplication:
```c
bool tquic_failover_is_duplicate(struct tquic_failover_ctx *fc, u64 pn)
{
    u8 idx;

    if (pn <= fc->highest_recv_pn - 256)
        return true;  /* Very old, assume duplicate */

    if (pn > fc->highest_recv_pn)
        return false;  /* New packet */

    idx = (u8)(pn & 0xFF);
    return test_bit(idx, fc->recv_bitmap);
}
EXPORT_SYMBOL_GPL(tquic_failover_is_duplicate);

void tquic_failover_mark_received(struct tquic_failover_ctx *fc, u64 pn)
{
    u8 idx;

    if (pn > fc->highest_recv_pn) {
        /* Clear bits for old packets falling out of window */
        while (fc->highest_recv_pn < pn) {
            idx = (u8)((fc->highest_recv_pn + 1) & 0xFF);
            clear_bit(idx, fc->recv_bitmap);
            fc->highest_recv_pn++;
        }
    }

    idx = (u8)(pn & 0xFF);
    if (test_and_set_bit(idx, fc->recv_bitmap))
        fc->stats.duplicates_received++;
}
EXPORT_SYMBOL_GPL(tquic_failover_mark_received);
```
