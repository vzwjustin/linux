Create the path validation state machine with adaptive timeouts:

1. In `include/net/tquic.h`, add to `struct tquic_path`:
   ```c
   /* Validation state */
   struct {
       u8 challenge_data[8];         /* Sent challenge */
       ktime_t challenge_sent;       /* When challenge was sent */
       bool challenge_pending;       /* Awaiting response */
       u8 retries;                   /* Retry count */
       struct timer_list timer;      /* Retransmission timer */
   } validation;

   /* Response queue (prevent memory exhaustion - pitfall #1) */
   struct {
       struct sk_buff_head queue;    /* Pending PATH_RESPONSE frames */
       atomic_t count;               /* Current queue depth */
   } response;
   #define TQUIC_MAX_PENDING_RESPONSES 256
   ```

2. Create `net/tquic/pm/path_validation.c` with:

   **Validation initiation** (`tquic_path_start_validation`):
   - Generate 8 random bytes: `get_random_bytes(path->validation.challenge_data, 8)`
   - Set state to TQUIC_PATH_PENDING
   - Reset retries to 0
   - Send PATH_CHALLENGE frame
   - Record challenge_sent time
   - Set challenge_pending = true
   - Start timer with initial timeout (1 second if no RTT, else 3x SRTT)

   **Retransmission timer callback** (`tquic_path_validation_timeout`):
   ```c
   if (path->validation.retries >= pernet->validation_retries) {
       /* Max retries exceeded - validation failed */
       path->state = TQUIC_PATH_FAILED;
       path->validation.challenge_pending = false;
       del_timer(&path->validation.timer);
       /* Emit TQUIC_PM_EVENT_FAILED */
       tquic_pm_path_event(conn, path, TQUIC_PM_EVENT_FAILED);
       /* Trigger failover */
       tquic_bond_path_failed(conn, path);
       return;
   }
   /* Retry */
   path->validation.retries++;
   tquic_path_send_challenge(conn, path);
   /* Reset timer with 3x SRTT (adaptive timeout) */
   u32 timeout_us = path->stats.rtt_smoothed ?
                    path->stats.rtt_smoothed * 3 :
                    1000000; /* 1 sec default */
   timeout_us = clamp(timeout_us, 100000U, 10000000U); /* 100ms - 10s */
   mod_timer(&path->validation.timer,
             jiffies + usecs_to_jiffies(timeout_us));
   ```

   **Challenge handling** (`tquic_path_handle_challenge`):
   - Check response queue depth (pitfall #1):
     ```c
     if (atomic_read(&path->response.count) >= TQUIC_MAX_PENDING_RESPONSES) {
         /* Drop to prevent memory exhaustion */
         return -ENOBUFS;
     }
     ```
   - Queue PATH_RESPONSE with received challenge data
   - Increment response count
   - Send response immediately or schedule

   **Response handling** (`tquic_path_handle_response`):
   - Match response data against sent challenge
   - If mismatch, log and return -EINVAL
   - Calculate RTT sample from challenge_sent to now
   - Update RTT statistics using RFC 6298 algorithm
   - Set path state to TQUIC_PATH_VALIDATED (or TQUIC_PATH_ACTIVE)
   - Clear challenge_pending
   - Stop timer
   - Emit TQUIC_PM_EVENT_VALIDATED

3. RTT update (leverage existing `tquic_pm_update_rtt` in path_manager.c or consolidate):
   - First sample: SRTT = sample, RTTVAR = sample/2
   - Subsequent: RFC 6298 with alpha=1/8, beta=1/4
   - Track min_rtt for scheduler
   - Validation timeout = SRTT + max(1ms, 4*RTTVAR), then multiply by 3

4. Timer initialization in path creation, cleanup in path destruction

5. Update Makefile

1. In `net/tquic/core/frame.c`:

   **Frame type constants** (if not already defined):
   ```c
   #define TQUIC_FRAME_PATH_CHALLENGE  0x1a
   #define TQUIC_FRAME_PATH_RESPONSE   0x1b
   ```

   **Encoding** (`tquic_frame_encode_path_challenge`):
   ```c
   int tquic_frame_encode_path_challenge(struct sk_buff *skb,
                                          const u8 *data)
   {
       /* Frame type (1 byte) + data (8 bytes) = 9 bytes total */
       if (skb_tailroom(skb) < 9)
           return -ENOBUFS;

       skb_put_u8(skb, TQUIC_FRAME_PATH_CHALLENGE);
       skb_put_data(skb, data, 8);
       return 9;
   }
   ```

   **Encoding** (`tquic_frame_encode_path_response`):
   ```c
   int tquic_frame_encode_path_response(struct sk_buff *skb,
                                         const u8 *data)
   {
       if (skb_tailroom(skb) < 9)
           return -ENOBUFS;

       skb_put_u8(skb, TQUIC_FRAME_PATH_RESPONSE);
       skb_put_data(skb, data, 8);
       return 9;
   }
   ```

   **Decoding** (`tquic_frame_decode_path_challenge`):
   ```c
   int tquic_frame_decode_path_challenge(const u8 *data, size_t len,
                                          u8 *out_data)
   {
       /* Expect exactly 8 bytes after frame type */
       if (len < 8)
           return -EINVAL;

       memcpy(out_data, data, 8);
       return 8;
   }
   ```
   (Same pattern for PATH_RESPONSE)

2. In `net/tquic/tquic_input.c`, add frame dispatch:
   ```c
   case TQUIC_FRAME_PATH_CHALLENGE:
       ret = tquic_frame_decode_path_challenge(data + 1, remaining - 1,
                                                challenge_data);
       if (ret < 0)
           return ret;
       tquic_path_handle_challenge(conn, path, challenge_data);
       consumed = 1 + 8;
       break;

   case TQUIC_FRAME_PATH_RESPONSE:
       ret = tquic_frame_decode_path_response(data + 1, remaining - 1,
                                               response_data);
       if (ret < 0)
           return ret;
       tquic_path_handle_response(conn, path, response_data);
       consumed = 1 + 8;
       break;
   ```

3. Add frame sending helper:
   ```c
   int tquic_path_send_challenge(struct tquic_connection *conn,
                                  struct tquic_path *path)
   {
       struct sk_buff *skb;
       int ret;

       skb = tquic_alloc_skb(conn, 9);
       if (!skb)
           return -ENOMEM;

       ret = tquic_frame_encode_path_challenge(skb,
                   path->validation.challenge_data);
       if (ret < 0) {
           kfree_skb(skb);
           return ret;
       }

       /* Send on specific path */
       return tquic_xmit_skb(conn, skb, path);
   }
   ```
