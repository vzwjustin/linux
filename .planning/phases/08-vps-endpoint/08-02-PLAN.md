4. Create net/tquic/tquic_qos.c for QoS classification:
   ```c
   // Read router hint and apply to tunnel
   int tquic_qos_classify(struct tquic_tunnel *tunnel, u8 router_hint)
   {
       // Validate hint is in range 0-3
       if (router_hint > 3)
           router_hint = 2;  // Default to bulk

       tunnel->traffic_class = router_hint;

       // Optionally override based on port (e.g., force 443 to interactive)
       if (tunnel->dest_port == 5060 || tunnel->dest_port == 5061)
           tunnel->traffic_class = 0;  // SIP = realtime

       return tunnel->traffic_class;
   }

   // Mark skb with traffic class for tc/nftables
   void tquic_qos_mark_skb(struct sk_buff *skb, struct tquic_tunnel *tunnel)
   {
       // Set skb->priority for tc HTB class selection
       skb->priority = tunnel->traffic_class;

       // Set DSCP based on class (for external QoS)
       // 0=EF (46), 1=AF41 (34), 2=BE (0), 3=CS1 (8)
       static const u8 dscp_map[] = { 46, 34, 0, 8 };
       ip_change_dscp(skb, dscp_map[tunnel->traffic_class]);
   }
   ```

Create net/tquic/tquic_forward.c for zero-copy data forwarding with hairpin:

1. Implement splice-based forwarding:
   ```c
   int tquic_forward_splice(struct tquic_tunnel *tunnel, int direction)
   {
       struct pipe_inode_info *pipe;
       ssize_t spliced;

       // Create pipe for splice buffer
       pipe = alloc_pipe_info();

       if (direction == TQUIC_FORWARD_TX) {
           // QUIC stream -> pipe -> TCP socket
           spliced = splice_from_quic(tunnel->quic_stream, pipe);
           splice_to_socket(pipe, tunnel->tcp_sock, spliced);
       } else {
           // TCP socket -> pipe -> QUIC stream
           spliced = splice_from_socket(tunnel->tcp_sock, pipe);
           splice_to_quic(pipe, tunnel->quic_stream, spliced);
       }

       // Apply QoS marking to outbound packets
       if (direction == TQUIC_FORWARD_TX)
           tquic_qos_mark_skb(current_skb, tunnel);

       return spliced;
   }
   ```

2. Use kernel splice internals (not syscall):
   - `do_splice()` or `splice_direct_to_actor()` for kernel-to-kernel
   - SPLICE_F_MOVE flag for zero-copy
   - Handle partial splices (loop until complete or would-block)

3. Implement hairpin traffic detection:
   ```c
   // Check if destination is another connected client
   struct tquic_client *tquic_forward_check_hairpin(struct tquic_tunnel *tunnel)
   {
       struct tquic_client *peer;

       // Check if dest IP falls within any client's port range
       // (Port range indicates traffic from that client)
       list_for_each_entry(peer, &tquic_client_list, list) {
           if (peer == tunnel->client)
               continue;  // Skip self

           // Check if dest is in peer's allocated address/port range
           if (tquic_client_owns_address(peer, tunnel->dest_addr, tunnel->dest_port))
               return peer;  // Hairpin to this client
       }
       return NULL;  // Normal forward to internet
   }

   // Route hairpin traffic to peer client's stream
   int tquic_forward_hairpin(struct tquic_tunnel *tunnel, struct tquic_client *peer)
   {
       // Find peer's connection and create reverse stream
       struct tquic_connection *peer_conn = tquic_client_get_connection(peer);
       if (!peer_conn)
           return -ENOENT;

       // Forward data directly to peer's stream (no TCP socket needed)
       return tquic_stream_forward(tunnel->quic_stream, peer_conn);
   }
   ```

4. Integrate with socket receive callback:
   - Hook into TCP sk_data_ready callback for RX
   - Hook into QUIC stream receive for TX
   - Use workqueue for async forwarding (avoid blocking in softirq)
   - Check for hairpin before creating TCP socket

5. Add MTU-aware segmentation:
   - Track per-path MTU (per CONTEXT.md)
   - Segment outbound QUIC packets to path MTU
   - Use TCP socket's MSS for TCP side
