   // Subscribe to kernel connection events via netlink multicast
   func (c *Client) SubscribeConnectionEvents(handler func(ConnectionEvent)) error {
       // Join TQUIC netlink multicast group for connection events
       // Events emitted by kernel on: connect, disconnect, migration
       // TQUIC_EVENT_CONN_OPEN, TQUIC_EVENT_CONN_CLOSE, TQUIC_EVENT_CONN_MIGRATE
   }

   type ConnectionEvent struct {
       Type       int       // OPEN, CLOSE, MIGRATE
       ClientID   string
       SourceIP   net.IP
       SourcePort uint16
       DestIP     net.IP

4. Create monitor/logger.go for detailed connection logging:
   ```go
   type ConnectionLogger struct {
       syslog    *syslog.Writer
       file      *os.File
   }

   type ConnectionLogEntry struct {
       Timestamp    time.Time `json:"timestamp"`
       ClientID     string    `json:"client_id"`
       SourceIP     string    `json:"source_ip"`
       SourcePort   uint16    `json:"source_port"`
       DestIP       string    `json:"dest_ip"`
       DestPort     uint16    `json:"dest_port"`
       BytesTx      uint64    `json:"bytes_tx"`
       BytesRx      uint64    `json:"bytes_rx"`
       DurationMs   uint64    `json:"duration_ms"`
       TrafficClass string    `json:"traffic_class"`  // "realtime", "interactive", etc.
       Action       string    `json:"action"`         // "open", "close", "migrate"
   }

   func (l *ConnectionLogger) LogConnection(event ConnectionEvent) {
       entry := ConnectionLogEntry{
           Timestamp:    event.Timestamp,
           ClientID:     event.ClientID,
           SourceIP:     event.SourceIP.String(),
           SourcePort:   event.SourcePort,
           DestIP:       event.DestIP.String(),
           DestPort:     event.DestPort,
           BytesTx:      event.BytesTx,
           BytesRx:      event.BytesRx,
           DurationMs:   event.DurationMs,
           TrafficClass: trafficClassNames[event.TrafficClass],
           Action:       eventTypeNames[event.Type],
       }

       // Write to syslog (structured message)
       l.syslog.Info(fmt.Sprintf("TQUIC %s: %s -> %s:%d, tx=%d rx=%d dur=%dms",
           entry.Action, entry.ClientID, entry.DestIP, entry.DestPort,
           entry.BytesTx, entry.BytesRx, entry.DurationMs))

       // Write JSON to dedicated log file
       json.NewEncoder(l.file).Encode(entry)
   }
   ```
